# Ability, Intent, and Capability

This page unifies three high-frequency modal patterns: 会, 能, and 要.

<CharacterGrid
    characters={[
        { char: '会', pinyin: 'huì', english: 'know how to; can (learned ability)' },
        { char: '能', pinyin: 'néng', english: 'be able to; can (capacity/condition)' },
        { char: '要', pinyin: 'yào', english: 'to want / going to' },
    ]}
/>

<RuleChecklist
    title="Core Rules"
    rules={[
        {
            rule: 'Use 会 for acquired skill.',
            right: '我会游泳。'
        },
        {
            rule: 'Use 能 for capability in current conditions.',
            right: '我今天不能跑步。'
        },
        {
            rule: 'Use 要 for plan/intention; 会 for softer prediction.',
            right: '明天我女朋友要来我家。 / 明天会下雨。'
        }
    ]}
/>

<PatternBox
    title="Contrast Pattern"
    pattern={[
        { part: '我', label: 'I' },
        { part: '会', label: 'skill' },

        { part: '但是', label: 'but' },
        { part: '我', label: 'I' },
        { part: '不能', label: 'current capability limit' },

    }}
/>
