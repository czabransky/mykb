# Ongoing Actions and Natural Progressive

This concept focuses on asking and answering about actions in progress.

<RuleChecklist
    title="Core Rules"
    rules={[
        {
            rule: 'Use 在 before a verb to show an action in progress.',
            right: '我在工作。'
        },
        {
            rule: 'Use 呢 optionally to keep a conversational tone.',
            right: '我在做饭呢。'
        },
        {
            rule: 'Do not use 在 + 去 for planned future movement.',
            why: 'For planned movement, 要去 is usually better.',
            right: '我们要去多伦多。',
            wrong: '我们在去多伦多呢。'
        }
    ]}
/>

<PatternBox
    title="Question + Answer"
    pattern={[
        { part: '你', label: 'you' },
        { part: '在', label: 'ongoing marker' },
        { part: '做什么', label: 'do what' },
        { part: '？', label: 'question' },
    ]}
    example={{
        chinese: '你在做什么？我在学中文呢。',
        pinyin: 'nǐ zài zuò shénme? wǒ zài xué Zhōngwén ne.',
        english: 'What are you doing? I am studying Chinese.'
    }}
/>
